  else UI.pageReady();

  document.getElementById("log").textContent =
    "fn=" + fn +
    " | loading.hidden=" + document.getElementById("loading").hidden +
    " content.hidden=" + document.getElementById("content").hidden +
    " errorBox.hidden=" + document.getElementById("errorBox").hidden +
    " boxA.hidden=" + document.getElementById("boxA").hidden;
})();
